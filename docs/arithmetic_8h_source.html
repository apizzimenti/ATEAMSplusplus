27 template <typename RingLike>
28 inline void SparseVectorAddition(
29 SparseVector<RingLike>& u,
30 const SparseVector<RingLike>& v,
31 Ring* R,
32 ComputeResources<RingLike>& resources
33 ) {
34 // If one of the vectors is zero, just return.
35 if (v.nnz() == 0) return;
36 if (u.nnz() == 0) return;
37
38 // Start from the largest index of each vector.
39 size_t left = u.nnz();
40 size_t right = v.nnz();
41 size_t out = u.nnz() + v.nnz();
42
43 // Preallocate enough space to accommodate all entries in both vectors.
44 if (u.alloc() < out) u.reserve(out);
45 u.resize(out);
46
47 while (left > 0 && right > 0) {
48 // If we're pointing to the same index...
49 if (u(left-1) == v(right-1)) {
50 auto entry = resources.arithmetic->add[(int)u[left-1]][(int)v[right-1]];
51
52 // ... and we get a nonzero entry, then the index of this entry
53 // is the same as in the original vector, and the entry is the
54 // result of the arithmetic; we then decrement the `out` pointer...
55 if (entry != 0) {
56 u(out-1) = u(left-1);
57 u[out-1] = entry;
58 out--;
59 }
60
61 // ... and decrement both left and right pointers, since we
62 // found a result in both.
63 left--;
64 right--;
65 }
66
67 // If we *aren't* pointing at the same index, and the index on the
68 // left is smaller than the index on the right...
69 else if (u(left-1) < v(right-1)) {
70 // ... then we set the index of this entry in the output vector
71 // to the index of the one on the right (since this preserves
72 // the order), then set the entry...
73 u(out-1) = v(right-1);
74 u[out-1] = v[right-1];
75
76 // ... then decrement the right pointer and the out pointer,
77 // since they track together in this case.
78 right--;
79 out--;
80 }
81
82 // Otherwise, the index on the right is smaller than the one on
83 // the left, and we do the same thing as the "else if" from above
84 // but with the roles reversed.
85 else {
86 u(out-1) = u(left-1);
87 u[out-1] = u[left-1];
88 left--;
89 out--;
90 }
91 }
92
93 // Then, while the right pointer is still bigger than zero, write
94 // the remaining entries into the vector.
95 while (right > 0) {
96 u(out-1) = v(right-1);
97 u[out-1] = v[right-1];
98 right--;
99 out--;
100 }
101
102 // Zero out the remaining entries, if there are any.
103 // #pragma omp parallel for default(shared) schedule(guided)
104 for (size_t i=left; i < out; i++) u[i] = 0;
105
106 // Canonicalize.
107 u.canonicalize();
108 }
