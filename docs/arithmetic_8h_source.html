47 template <typename RingLike>
48 inline void SparseVectorAddition(
49 SparseVector<RingLike>& u,
50 SparseVector<RingLike>& v,
51 Ring* R,
52 ComputeResources<RingLike>& resources
53 ) {
54 // If one of the vectors is zero, just return.
55 if (v.nnz() == 0) return;
56 if (u.nnz() == 0) return;
57
58 // Start from the largest index of each vector.
59 size_t left = u.nnz();
60 size_t right = v.nnz();
61 size_t out = u.nnz() + v.nnz();
62
63 // Preallocate enough space to accommodate all entries in both vectors.
64 if (u.alloc() < out) u.reserve(out);
65 u.resize(out);
66
67 while (left > 0 && right > 0) {
68 // If we're pointing to the same index...
69 if (u(left-1) == v(right-1)) {
70 auto entry = resources.arithmetic->add[u[left-1]][v[right-1]];
71
72 // ... and we get a nonzero entry, then the index of this entry
73 // is the same as in the original vector, and the entry is the
74 // result of the arithmetic; we then decrement the `out` pointer...
75 if (entry != 0) {
76 u(out-1) = u(left-1);
77 u[out-1] = entry;
78 out--;
79 }
80
81 // ... and decrement both left and right pointers, since we
82 // found a result in both.
83 left--;
84 right--;
85 }
86
87 // If we *aren't* pointing at the same index, and the index on the
88 // left is smaller than the index on the right...
89 else if (u(left-1) < v(right-1)) {
90 // ... then we set the index of this entry in the output vector
91 // to the index of the one on the right (since this preserves
92 // the order), then set the entry...
93 u(out-1) = v(right-1);
94 u[out-1] = v[right-1];
95
96 // ... then decrement the right pointer and the out pointer,
97 // since they track together in this case.
98 right--;
99 out--;
100 }
101
102 // Otherwise, the index on the right is smaller than the one on
103 // the left, and we do the same thing as the "else if" from above
104 // but with the roles reversed.
105 else {
106 u(out-1) = u(left-1);
107 u[out-1] = u[left-1];
108 left--;
109 out--;
110 }
111 }
112
113 // Then, while the right pointer is still bigger than zero, write
114 // the remaining entries into the vector.
115 while (right > 0) {
116 u(out-1) = v(right-1);
117 u[out-1] = v[right-1];
118 right--;
119 out--;
120 }
121
122 // Zero out the remaining entries, if there are any.
123 // #pragma omp parallel for default(shared) schedule(guided)
124 for (size_t i=left; i < out; i++) u[i] = 0;
125
126 // Canonicalize.
127 u.canonicalize();
128 }

183 template <typename RingLike>
184 inline SparseVector<RingLike> OpenMPSparseRightMultiplication(
185 SparseMatrix<RingLike>& A,
186 SparseVector<RingLike>& x,
187 Ring* R
188 ) {
189 std::vector<SparseVector<RingLike>> chunks(omp_get_max_threads(), SparseVector<RingLike>());
190
191 // Compute the dot products in parallel.
192 #pragma omp parallel for default(shared) schedule(guided)
193 for (int i=0; i < A.rows.size(); i++) {
194 int tid = omp_get_thread_num();
195 typename RingLike::dtype r = arithmetic::SparseVectorDot<RingLike>(A.rows[i], x, R);
196
197 if (r > 0) chunks[tid].push_back((INDEX)i, r);
198 }
199
200 // Reconstitute the vector.
201 SparseVector<RingLike> r;
202
203 for (int j=0; j < chunks.size(); j++) {
204 // Skip this row if there's nothing in it.
205 if (chunks[j].size() < 1) continue;
206
207 for (int k=0; k < chunks[j].size(); k++) {
208 r.push_back(chunks[j](k), chunks[j][k]);
209 }
210 }
211 r.sort_indices();
212
213 return r;
214 }

230 template <typename RingLike>
231 inline SparseVector<RingLike> SparseRightMultiplication(
232 SparseMatrix<RingLike>& A,
233 SparseVector<RingLike>& x,
234 Ring* R,
235 ComputeResources<RingLike>& resources
236 ) {
237 // If parallelism isn't enabled, kick to the other version.
238 if (!resources.parallel->enabled) return SparseRightMultiplication<RingLike>(A, x, R);
239
240 // If we haven't yet decided on a preferred number of blocks, let's
241 // compute that now.
242 if (resources.typed->blocks < 0) {
243 resources.typed->defaultPreferredBlocks(A.rows.size());
244 resources.typed->build();
245 }
246
247
248 // Compute the dot products in parallel.
249 resources.opt->pool.detach_blocks(
250 0, A.rows.size(),
251 [&](const size_t start, const size_t end) {
252 if (resources.opt->abort) return;
253
254 // Get the thread ID
255 auto tid = SparseRREF::thread_id();
256 typename RingLike::dtype r;
257
258 // Loop over each block.
259 for (size_t i=start; i < end; i++) {
260 r = arithmetic::SparseVectorDot<RingLike>(A.rows[i], x, R);
261 if (r > 0) resources.typed->parts[tid].push_back((INDEX)i, r);
262 }
263 },
264 resources.typed->blocks
265 );
266 resources.opt->pool.wait();
267
268 // Reconstitute the vector.
269 SparseVector<RingLike> r;
270
271 for (int j=0; j < resources.typed->parts.size(); j++) {
272 // Skip this row if there's nothing in it.
273 if (resources.typed->parts[j].size() < 1) continue;
274
275 for (int k=0; k < resources.typed->parts[j].size(); k++) {
276 r.push_back(resources.typed->parts[j](k), resources.typed->parts[j][k]);
277 }
278 }
279 r.sort_indices();
280 resources.typed->flush();
281
282 return r;
283 }
