446 inline vector<int> standardRestrictedReportingPolicy(
447 complexes::Complex<RingLike>* complex,
448 vector<int>& lookup,
449 set<int>& marked,
450 int dimension
451 ) {
452 int low = complex->Breaks[dimension][0];
453 int high = complex->Breaks[dimension][1];
454
455 vector<int> essential;
456
457 for (int k : marked) {
458 if (lookup[k] == 0 && (low <= k && k < high)) essential.push_back(k);
459 }
460
461 return essential;
462 }
