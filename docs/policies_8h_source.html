384 complexes::Complex<RingLike>* complex,
385 vector<int>& lookup,
386 set<int>& marked,
387 int dimension
388 ) {
389 int low = complex->Breaks[dimension][0];
390 int high = complex->Breaks[dimension][1];
391
392 vector<int> essential;
393
394 for (int k : marked) {
395 if (lookup[k] == 0 && (low <= k && k < high)) essential.push_back(k);
396 }
