190 using ReportingPolicy = function<vector<int>(complexes::Complex<RingLike>*,vector<int>&,set<int>&)>;
191
192 template <typename RingLike>
193 inline vector<int> standardRestrictedReportingPolicy(
194 complexes::Complex<RingLike>* complex,
195 vector<int>& lookup,
196 set<int>& marked,
197 int dimension
198 ) {
199 int low = complex->Breaks[dimension][0];
200 int high = complex->Breaks[dimension][1];
201
202 vector<int> essential;
203
204 for (int k : marked) {
205 if (lookup[k] == 0 && (low <= k && k < high)) essential.push_back(k);
206 }
207
208 return essential;
209 }
