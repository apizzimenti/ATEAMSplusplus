20 template <typename RingLike>
21 inline void reduceChain(
22 SparseMatrix<RingLike>& Full,
23 vector<int>& lookup,
24 SparseVector<RingLike>& cell,
25 Ring* R
26 ) {
27 // Get the youngest chain (column) sharing a face with `cell` and zero
28 // out the pivot row.
29 SparseVector<RingLike> youngest = Full.rows[lookup[helpers::youngestOf<RingLike>(cell)]];
30
31 typename RingLike::dtype q = *youngest.find(helpers::youngestOf<RingLike>(cell));
32 typename RingLike::dtype s = scalar_neg(scalar_inv(q, R->ring), R->ring);
33
34 arithmetic::SparseVectorRescaling<RingLike>(s, youngest, R);
35 arithmetic::SparseVectorAddition<RingLike>(cell, youngest, R);
36 }

39 template <typename RingLike>
40 inline void reduceBlock(
41 SparseMatrix<RingLike>& Full,
42 vector<int>& endpoints,
43 vector<int>& lookup,
44 int dim,
45 Ring *R,
46 ReductionPolicy<RingLike> reductionPolicy,
47 CreationPolicy creationPolicy,
48 DestructionPolicy<RingLike> destructionPolicy,
49 arithmetic::ComputeOptions<RingLike>& options
50 ) {
51 for (int j=endpoints[0]; j < endpoints[1]; j++) {
52 SparseVector<RingLike>& cell = Full.rows[j];
53
54 while (reductionPolicy(cell, lookup, j, dim)) reduceChain<RingLike>(Full, lookup, cell, R);
55
56 if (cell.size() > 0) destructionPolicy(cell, j, dim);
57 else creationPolicy(j, dim);
58 }
59 }
60}

67 inline vector<int> standard(
68 complexes::Complex<RingLike>* complex,
69 vector<int>& filtration,
70 Ring* R,
71 arithmetic::ComputeOptions<RingLike>& options,
72 ReindexingPolicy<RingLike> reindexingPolicy,
73 TraversalPolicy<RingLike> traversalPolicy,
74 ReportingPolicy<RingLike> reportingPolicy
75 ) {
76 // Determine the endpoints and reindex the boundary matrix accordingly.
77 SparseMatrix<RingLike> Full = reindexingPolicy(complex, filtration, options);
78
79 // Track which column is to be added next; track which ones are marked.
80 vector<int> youngestChainLookup(complex->size(), 0);
81 set<int> marked;
82
83 // Cycle creation policy.
84 auto creationPolicy = std::bind(
85 standardCreationPolicy<RingLike>, // using the standard creation policy
86 placeholders::_1, // placeholder for `markedIndex`
87 placeholders::_2, // placeholder for `dim`
88 std::ref(marked) // a reference to `marked`.
89 );
90
91 // Cycle destruction policy.
92 auto destructionPolicy = std::bind(
93 standardDestructionPolicy<RingLike>, // using the standard destruction policy
94 placeholders::_1, // placeholder for `cell`
95 placeholders::_2, // placeholder for `markedIndex`
96 placeholders::_3, // placeholder for `dim`,
97 std::ref(youngestChainLookup) // reference to `youngestChainLookup`.
98 );
99
100 // Reduce the blocks of the matrix.
101 vector<int> endpoints = traversalPolicy(complex);
102
103 for (int d=endpoints[0]; d <= endpoints[1]; d++) {
104 reduceBlock<RingLike>(
105 Full,
106 complex->Breaks[d],
107 youngestChainLookup,
108 d,
109 R,
110 standardReductionPolicy<RingLike>,
111 creationPolicy,
112 destructionPolicy,
113 options
114 );
115 }
116
117 // Find essential cycles.
118 return reportingPolicy(
119 complex,
120 youngestChainLookup,

583 template <typename RingLike>
584 inline vector<int> PHAT(
585 complexes::Complex<RingLike>* complex,
586 vector<int>& filtration
587 ) {
588 // The filtration specifies the order in which we add the cells of all
589 // dimensions. Create a map that specifies to which position each cell was
590 // moved. For example, if the filtration has
591 //
592 // [... 12, 9, 10, 19, ...]
593 //
594 // that was originally
595 // [... 9, 10, 11, 12, ...]
596 //
597 // then the mapping should have entries
598 //
599 // {... 12: 9, 9: 10, 10: 11, 19: 12, ...}
600
601 // TODO optimize, this takes way too long
602 vector<int> remapping(filtration.size(), 0);
603 for (int t=0; t < filtration.size(); t++) remapping[filtration[t]] = t;
604
605 PHATBoundaryMatrix boundary;
606 PHATColumn column;
607
608 vector<int> ground;
609 int dim;
610
611 boundary.set_num_cols(complex->Boundary.Flat.size());
612
613 // this also takes an insane amount of time; no need to go all the way
614 // up.
615 for (int t=0; t < complex->Boundary.Flat.size(); t++) {
616 ground = complex->Boundary.Flat[filtration[t]];
617
618 for (int j=0; j < ground.size(); j++) {
619 column.push_back(remapping[ground[j]]);
620 }
621
622 std::sort(column.begin(), column.end());
623
624 // Determine the dimension of the cell.
625 for (int d=0; d < complex->Breaks.size(); d++) {
626 int lo = complex->Breaks[d][0];
627 int hi = complex->Breaks[d][1];
628
629 if (lo <= filtration[t] && filtration[t] < hi) {
630 dim = d;
631 break;
632 }
633 }
634
635 boundary.set_dim(t, dim);
636 boundary.set_col(t, column);
637 column.clear();
638 }
639
640 // Compute the persistence pairs and populate a Vector to return to the
641 // user. We know when the giant(/essential) cycles are born because they
642 // do not appear in the list of births or deaths.
643 PHATPairs pairs;
644 set<int> all;
645 phat::compute_persistence_pairs<PHATTwist>(pairs, boundary);
646 pairs.sort();
647
648 for (phat::index i=0; i < pairs.get_num_pairs(); i++) {
649 all.insert(pairs.get_pair(i).first);
650 all.insert(pairs.get_pair(i).second);
651 }
652
653 vector<int> essential;
654
655 for (int t=0; t < filtration.size(); t++) {
656 if (!all.contains(filtration[t])) essential.push_back(filtration[t]);
657 }
658
659 return essential;
660 }

664 template <typename RingLike>
665 inline vector<int> PHAT(
666 complexes::Complex<RingLike>* complex,
667 vector<int>& filtration,
668 int dimension
669 ) {
670 // No sense in doing a ton of policy work for this one.
671 vector<int> essential = PHAT<RingLike>(complex, filtration);
672 int low = complex->Breaks[dimension][0], high = complex->Breaks[dimension][1];
673 std::erase_if(essential, [filtration,low,high](int t) { return !(low <= filtration[t] && filtration[t] < high); });
674
675 return essential;
676 }

686 ) {
