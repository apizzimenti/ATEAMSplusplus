150 template <typename RingLike>
151 struct FiniteArithmetic {
152 std::vector<std::vector<typename RingLike::dtype>> add;
153 std::vector<std::vector<typename RingLike::dtype>> multiply;
154 std::vector<typename RingLike::dtype> negate;
155 std::vector<typename RingLike::dtype> invert;
156
157 void arithmetize(Ring* R) {
158 // Create addition table.
159 int c = R->characteristic;
160 std::vector<std::vector<typename RingLike::dtype>> addition(c, std::vector<typename RingLike::dtype>(c, 0));
161
162 for (int a=0; a < c; a++) {
163 for (int b=0; b < c; b++) {
164 addition[a][b] = (typename RingLike::dtype)((a+b) % c);
165 }
166 }
167
168 // Create multiplication table.
169 std::vector<std::vector<typename RingLike::dtype>> multiplication(c, std::vector<typename RingLike::dtype>(c, 0));
170 for (int a=0; a < c; a++) {
171 for (int b=0; b < c; b++) {
172 multiplication[a][b] = (typename RingLike::dtype)((a*b) % c);
173 }
174 }
175
176 // Create negation table.
177 std::vector<typename RingLike::dtype> negation(c, 0);
178 for (int a=0; a < c; a++) {
179 negation[a] = (typename RingLike::dtype)((c-a)%c);
180 }
181
182 // Create inversion table. Fermat's little theorem!
183 std::vector<typename RingLike::dtype> inversion(c, 0);
184 for (int a=0; a < c; a++) {
185 inversion[a] = (typename RingLike::dtype)(((int)pow(a, c-2))%c);
186 }
187
188 this->add = addition;
189 this->multiply = multiplication;
190 this->negate = negation;
191 this->invert = inversion;
192 };
193 };
