230 struct FiniteArithmetic {
231 std::vector<std::vector<typename RingLike::dtype>> add;
232 std::vector<std::vector<typename RingLike::dtype>> multiply;
233 std::vector<typename RingLike::dtype> negate;
234 std::vector<typename RingLike::dtype> invert;
235
241 void arithmetize(Ring* R) {
242 // Create addition table.
243 int c = R->characteristic;
244 std::vector<std::vector<typename RingLike::dtype>> addition(c, std::vector<typename RingLike::dtype>(c, 0));
245
246 for (int a=0; a < c; a++) {
247 for (int b=0; b < c; b++) {
248 addition[a][b] = (typename RingLike::dtype)((a+b) % c);
249 }
250 }
251
252 // Create multiplication table.
253 std::vector<std::vector<typename RingLike::dtype>> multiplication(c, std::vector<typename RingLike::dtype>(c, 0));
254 for (int a=0; a < c; a++) {
255 for (int b=0; b < c; b++) {
256 multiplication[a][b] = (typename RingLike::dtype)((a*b) % c);
257 }
258 }
259
260 // Create negation table.
261 std::vector<typename RingLike::dtype> negation(c, 0);
262 for (int a=0; a < c; a++) {
263 negation[a] = (typename RingLike::dtype)((c-a)%c);
264 }
265
266 // Create inversion table. Fermat's little theorem!
267 std::vector<typename RingLike::dtype> inversion(c, 0);
268 for (int a=0; a < c; a++) {
269 inversion[a] = (typename RingLike::dtype)(((int)pow(a, c-2))%c);
270 }
271
272 this->add = addition;
273 this->multiply = multiplication;
274 this->negate = negation;
275 this->invert = inversion;
276 };
