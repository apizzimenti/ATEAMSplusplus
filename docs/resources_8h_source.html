252 struct FiniteArithmetic {
253 std::vector<std::vector<typename RingLike::dtype>> add;
254 std::vector<std::vector<typename RingLike::dtype>> multiply;
255 std::vector<typename RingLike::dtype> negate;
256 std::vector<typename RingLike::dtype> invert;
257
263 void arithmetize(Ring* R) {
264 // Create addition table.
265 int c = R->characteristic;
266 std::vector<std::vector<typename RingLike::dtype>> addition(c, std::vector<typename RingLike::dtype>(c, 0));
267
268 for (int a=0; a < c; a++) {
269 for (int b=0; b < c; b++) {
270 addition[a][b] = (typename RingLike::dtype)((a+b) % c);
271 }
272 }
273
274 // Create multiplication table.
275 std::vector<std::vector<typename RingLike::dtype>> multiplication(c, std::vector<typename RingLike::dtype>(c, 0));
276 for (int a=0; a < c; a++) {
277 for (int b=0; b < c; b++) {
278 multiplication[a][b] = (typename RingLike::dtype)((a*b) % c);
279 }
280 }
281
282 // Create negation table.
283 std::vector<typename RingLike::dtype> negation(c, 0);
284 for (int a=0; a < c; a++) {
285 negation[a] = (typename RingLike::dtype)((c-a)%c);
286 }
287
288 // Create inversion table. Fermat's little theorem!
289 std::vector<typename RingLike::dtype> inversion(c, 0);
290 for (int a=0; a < c; a++) {
291 inversion[a] = (typename RingLike::dtype)(((int)pow(a, c-2))%c);
292 }
293
294 this->add = addition;
295 this->multiply = multiplication;
296 this->negate = negation;
297 this->invert = inversion;
298 };
