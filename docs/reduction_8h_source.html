14 template <typename RingLike>
15 inline void reduceChain(
16 SparseMatrix<RingLike>& Full,
17 std::vector<int>& lookup,
18 SparseVector<RingLike>& cell,
19 Ring* R
20 ) {
21 // Get the youngest chain (column) sharing a face with `cell` and zero
22 // out the pivot row.
23 SparseVector<RingLike> youngest = Full.rows[lookup[helpers::youngestOf<RingLike>(cell)]];
24
25 typename RingLike::dtype q = *youngest.find(helpers::youngestOf<RingLike>(cell));
26 typename RingLike::dtype s = scalar_neg(scalar_inv(q, R->ring), R->ring);
27
28 arithmetic::SparseVectorRescaling<RingLike>(s, youngest, R);
29 arithmetic::SparseVectorAddition<RingLike>(cell, youngest, R);
30 }
